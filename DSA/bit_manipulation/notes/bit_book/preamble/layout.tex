% --- PART ---
\titleformat{\part}[display]
  {\centering\normalfont\Huge\bfseries\color{primaryblue}}
  {\partname\ \thepart}
  {20pt}
  {\Huge\color{deepblue}}
  [\vspace{2ex}{\color{primaryblue}\hrule height 3pt}\vspace{1ex}{\color{accentorange}\hrule height 1pt}]

% --- CHAPTER ---
\titleformat{\chapter}[display]
  {\normalfont\huge\bfseries\color{primaryblue}}
  {\tikz{\node[circle, fill=primaryblue, text=white, minimum size=1.5cm,
               font=\Large\bfseries] {\thechapter};}\quad\chaptertitlename}
  {10pt}
  {\Huge\color{deepblue}}
  [\vspace{1ex}{\color{secondaryblue}\titlerule[2pt]}]
\titlespacing*{\chapter}{0pt}{-20pt}{30pt}

% --- SECTION (a theory block, a pattern, the summary) ---
\titleformat{\section}
  {\normalfont\Large\bfseries\color{primaryblue}}
  {\tikz{\filldraw[primaryblue] (0,0) rectangle (0.3,0.3);}\quad\thesection}
  {1em}{}
  [\vspace{0.5ex}{\color{midgray}\titlerule[1pt]}]

% --- SUBSECTION (a solved problem) ---
\titleformat{\subsection}
  {\normalfont\large\bfseries\color{deepblue}}
  {\thesubsection}{0.8em}{}

\setcounter{secnumdepth}{2}   % number down to subsections
\setcounter{tocdepth}{2}      % show chapters, sections, subsections

% --- Header / footer ---
% Left: short chapter tag; right: current section (never overlap).
\newif\ifdpmainmatter
\apptocmd{\mainmatter}{\global\dpmainmattertrue}{}{}% after \mainmatter's own page break, so the last front-matter page keeps its header
\renewcommand{\chaptermark}[1]{\markboth{#1}{#1}}
\renewcommand{\sectionmark}[1]{\markright{\ifnum\value{secnumdepth}>0 \thesection\quad\fi #1}}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\textcolor{primaryblue}{\small\textbf{\ifdpmainmatter Chapter~\thechapter\else\nouppercase{\leftmark}\fi}}}
\fancyhead[R]{\textcolor{primaryblue}{\small\textbf{\nouppercase{\rightmark}}}}
\fancyfoot[L]{\textcolor{darkgray}{\thepage}}
\fancyfoot[C]{\textcolor{darkgray}{\small\textit{Bit Manipulation --- From Foundations to World Finals}}}
\renewcommand{\headrulewidth}{2pt}
\renewcommand{\footrulewidth}{1pt}
\renewcommand{\headrule}{\hbox to\headwidth{\color{primaryblue}\leaders\hrule height \headrulewidth\hfill}}
\renewcommand{\footrule}{\hbox to\headwidth{\color{lightgray}\leaders\hrule height \footrulewidth\hfill}}

% --- Links ---

    colorlinks=true,

    citecolor=accentorange,
    pdftitle={Bit Manipulation --- From Foundations to World Finals},

}

% --- Contents ---
% The main table of contents (frontmatter) lists parts, chapters and sections only;
% every chapter opens with its own detailed contents box (\chaptercontents).
\newcommand{\booktableofcontents}{\etocsetnexttocdepth{section}\tableofcontents}
\newcommand{\chaptercontents}{%
  \par\medskip
  \begin{tcolorbox}[enhanced, colback=white, colframe=primaryblue!70, boxrule=0.8pt, arc=3pt,
      left=6pt, right=6pt, top=4pt, bottom=2pt,
      title={\bfseries In This Chapter}, colbacktitle=primaryblue, coltitle=white, fonttitle=\small]
  \footnotesize
  \etocsettocstyle{\begin{multicols}{2}\raggedcolumns}{\end{multicols}}%
  \etocsetnexttocdepth{subsection}%
  \localtableofcontents
  \end{tcolorbox}%
}
