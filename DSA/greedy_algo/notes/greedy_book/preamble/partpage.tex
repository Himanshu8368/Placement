%============================================================
% partpage.tex — full-page part openers
%
%   \dppartopener{<title>}{<level range>}{<colour>}{<tagline>}
%       Starts a new page, registers the part in the table of contents and
%       PDF bookmarks, and draws the coloured title band at the top.
%   \dppartstats{<chapters>}{<patterns>}{<solved>}{<practice>}
%       A strip of four numbers.
%   \begin{dpjourney} ... \end{dpjourney}  with  \dpstep{<n>}{<title>}{<what it teaches>}{<level>}{<done: yes/no>}
%       The vertical chapter-by-chapter journey.
%   \dppartgoals{<items>}
%       "By the end of this part" box (items are \item ...).
%============================================================

\newcommand{\dppartcolor}{primaryblue}

\newcommand{\dppartopener}[4]{%
  \clearpage
  \thispagestyle{empty}%
  \renewcommand{\dppartcolor}{#3}%
  \refstepcounter{part}%
  \phantomsection
  \addcontentsline{toc}{part}{\thepart\hspace{1em}#1}%
  \begin{tikzpicture}[remember picture, overlay]
    % coloured band
    \fill[#3] (current page.north west) rectangle ([yshift=-7.2cm]current page.north east);
    \fill[#3!70!black] ([yshift=-7.2cm]current page.north west) rectangle ([yshift=-7.5cm]current page.north east);
    % large faint numeral
    \node[anchor=north east, font=\fontsize{120}{120}\selectfont\bfseries, text=white, opacity=0.15]
      at ([xshift=-1.2cm, yshift=-0.9cm]current page.north east) {\Roman{part}};
    % texts
    \node[anchor=north west, font=\Large\bfseries\scshape, text=white]
      at ([xshift=2.2cm, yshift=-1.6cm]current page.north west) {Part \Roman{part}};
    \node[anchor=north west, font=\fontsize{34}{38}\selectfont\bfseries, text=white]
      at ([xshift=2.2cm, yshift=-2.5cm]current page.north west) {#1};
    \node[anchor=north west, font=\large\itshape, text=white, text width=12cm]
      at ([xshift=2.2cm, yshift=-4.2cm]current page.north west) {#4};
    \node[anchor=north west, rectangle, rounded corners=8pt, fill=white, text=#3!80!black,
          font=\small\bfseries, inner xsep=10pt, inner ysep=5pt]
      at ([xshift=2.2cm, yshift=-5.6cm]current page.north west) {Level: #2};
  \end{tikzpicture}%
  \vspace*{4.8cm}%
}

\newcommand{\dppartstats}[4]{%
  \begin{center}
  \begin{tikzpicture}[stat/.style={rectangle, rounded corners=4pt, draw=\dppartcolor!60, fill=\dppartcolor!8,
                                   line width=0.8pt, minimum width=3.15cm, minimum height=1.5cm, align=center}]
    \node[stat] at (0,0)   {{\LARGE\bfseries\color{\dppartcolor!80!black}#1}\\[1pt]{\scriptsize\color{darkgray}chapters}};
    \node[stat] at (3.45,0) {{\LARGE\bfseries\color{\dppartcolor!80!black}#2}\\[1pt]{\scriptsize\color{darkgray}patterns}};
    \node[stat] at (6.9,0)  {{\LARGE\bfseries\color{\dppartcolor!80!black}#3}\\[1pt]{\scriptsize\color{darkgray}fully solved problems}};
    \node[stat] at (10.35,0) {{\LARGE\bfseries\color{\dppartcolor!80!black}#4}\\[1pt]{\scriptsize\color{darkgray}practice problems}};
  \end{tikzpicture}
  \end{center}
}

% chapter opener facts: \chapterfacts{<level from>}{<level to>}{<patterns>}{<solved>}{<practice>}
% a strip of four tiles in the chapter colour scheme (same look as \dppartstats)
\newcommand{\chapterfacts}[5]{%
  \begin{center}
  \begin{tikzpicture}[fact/.style={rectangle, rounded corners=4pt, draw=primaryblue!55, fill=primaryblue!6,
                                   line width=0.8pt, text width=2.95cm, minimum height=1.5cm, align=center}]
    \node[fact] at (0,0)     {\rule{0pt}{14pt}{\small\bfseries\color{deepblue}\mbox{#1 $\to$ #2}}\\[3pt]{\scriptsize\color{darkgray}level (Codeforces)}};
    \node[fact] at (3.45,0)  {{\LARGE\bfseries\color{deepblue}#3}\\[1pt]{\scriptsize\color{darkgray}patterns}};
    \node[fact] at (6.9,0)   {{\LARGE\bfseries\color{deepblue}#4}\\[1pt]{\scriptsize\color{darkgray}fully solved problems}};
    \node[fact] at (10.35,0) {{\LARGE\bfseries\color{deepblue}#5}\\[1pt]{\scriptsize\color{darkgray}practice problems}};
  \end{tikzpicture}
  \end{center}
}

% the journey: one row per chapter, drawn as a vertical timeline
\newcounter{dpstepcount}
\newenvironment{dpjourney}{%
  \setcounter{dpstepcount}{0}%
  \par\medskip\noindent{\large\bfseries\color{\dppartcolor!80!black} The Journey}\hfill
  {\scriptsize\color{darkgray}filled disc = written \quad dashed disc = planned}\par\smallskip
}{\par}

\newcommand{\dpstep}[5]{%
  \stepcounter{dpstepcount}%
  \noindent
  \begin{tikzpicture}
    \def\dpdone{#5}%
    \def\dpyes{yes}%
    % rail
    \draw[\dppartcolor!45, line width=2.2pt] (0,0.68) -- (0,-0.68);
    % chapter disc
    \ifx\dpdone\dpyes
      \node[circle, fill=\dppartcolor, text=white, minimum size=1.05cm, font=\large\bfseries] at (0,0) {#1};
    \else
      \node[circle, fill=white, draw=\dppartcolor, line width=1.2pt, dashed, text=\dppartcolor, minimum size=1.05cm, font=\large\bfseries] at (0,0) {#1};
    \fi
    % card
    \node[anchor=west, rectangle, rounded corners=4pt, draw=\dppartcolor!35, fill=white,
          line width=0.6pt, minimum height=1.15cm, text width=13.1cm, inner sep=5pt] (card) at (0.85,0) {%
        {\bfseries\color{deepblue}#2}\hfill
        {\scriptsize\color{\dppartcolor!80!black}\bfseries #4}\\[2pt]
        {\small\color{darkgray}#3}};
  \end{tikzpicture}\par\vspace{-1.5pt}%
}

\newcommand{\dppartgoals}[1]{%
  \par\medskip
  \begin{tcolorbox}[enhanced, colback=\dppartcolor!6, colframe=\dppartcolor, boxrule=0.8pt, arc=3pt,
                    left=8pt, right=8pt, top=4pt, bottom=4pt,
                    title={\bfseries By the end of this part you will be able to}, fonttitle=\small,
                    colbacktitle=\dppartcolor, coltitle=white]
  \small
  \begin{itemize}[leftmargin=*, itemsep=1pt, topsep=1pt]
  #1
  \end{itemize}
  \end{tcolorbox}
}
