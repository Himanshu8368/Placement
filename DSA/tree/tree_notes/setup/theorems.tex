% =====================================================================
%  setup/theorems.tex -- numbered boxes. All share ONE counter that is
%  reset in every topic (section), so numbers read 1.3.1, 1.3.2, ...
%  in reading order no matter which kind of box it is.
%
%  Usage (two mandatory arguments: description, label suffix):
%    \begin{theorem}{Edges of a Tree}{c01-edges-nodes}  ...  \end{theorem}
%        -> "Theorem 1.3.4 (Edges of a Tree)", label thm:c01-edges-nodes
%    \begin{definition}{Tree}{c01-tree} ... \end{definition}
%    \begin{lemma}{}{c01-step} ... \end{lemma}        (empty description is fine)
%    \begin{theorem*}{Unnumbered}{} ... \end{theorem*} (starred = no number)
%
%  Label prefixes: def: thm: lem: cor: prop: ex:  (added automatically).
%  Never put a numbered box before the first \section of a chapter.
% =====================================================================

\tcbset{
  thmbase/.style={
    enhanced, breakable, arc=2.5mm, boxrule=1.1pt,
    left=7pt, right=7pt, top=5pt, bottom=5pt,
    fonttitle=\bfseries, coltitle=textondark,
    separator sign none, description delimiters parenthesis,
    description font=\bfseries,
    before upper={\setlength{\parskip}{0.45em}}
  },
}

\newtcbtheorem[number within=section]{definition}{Definition}%
  {thmbase, colback=defbg, colframe=primaryblue, colbacktitle=primaryblue}{def}

\newtcbtheorem[use counter from=definition]{theorem}{Theorem}%
  {thmbase, colback=theorembg, colframe=softpurple, colbacktitle=softpurple}{thm}

\newtcbtheorem[use counter from=definition]{lemma}{Lemma}%
  {thmbase, colback=theorembg!55!white, colframe=softpurple!85!black,
   colbacktitle=softpurple!85!black}{lem}

\newtcbtheorem[use counter from=definition]{corollary}{Corollary}%
  {thmbase, colback=theorembg!55!white, colframe=softpurple!70!white,
   colbacktitle=softpurple!70!white}{cor}

\newtcbtheorem[use counter from=definition]{proposition}{Proposition}%
  {thmbase, colback=theorembg!55!white, colframe=softpurple!85!black,
   colbacktitle=softpurple!85!black}{prop}

\newtcbtheorem[use counter from=definition]{example}{Example}%
  {thmbase, colback=tracebg, colframe=traceborder, colbacktitle=traceborder}{ex}
