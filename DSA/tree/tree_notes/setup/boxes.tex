% Forward motivation: \usedin{Chapter 13 (segment trees)}{answer subtree-sum queries fast}
\newcommand{\usedin}[2]{%
  \par\smallskip\noindent
  \begin{tcolorbox}[enhanced, breakable, colback=successgreen!7, colframe=successgreen,
      boxrule=0.9pt, arc=2mm, left=7pt, right=7pt, top=4pt, bottom=4pt]
    \small\textbf{\textcolor{successgreen!70!black}{Where this is used.}}\ #1\ \ --\ \ #2\par
  \end{tcolorbox}\par\smallskip}

% Two-way disambiguation for the classic confusions. Short contrasting phrases only;
% for long comparisons use comparisonbox with your own tabularx.
%   \vsbox{Depth vs Height}{Depth}{Height}{from the root down}{from a leaf up}
\newcommand{\vsbox}[5]{%
  \begin{comparisonbox}[#1]%
  \setlength{\tabcolsep}{6pt}\renewcommand{\arraystretch}{1.15}%
  \begin{tabular}{@{}p{0.455\linewidth}p{0.455\linewidth}@{}}
    \textbf{\textcolor{deepblue}#2} & \textbf{\textcolor{deepblue}#3}\\
    #4 & #5\\
  \end{tabular}
  \end{comparisonbox}}
