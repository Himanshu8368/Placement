% ---------------- Parts ----------------
\newcommand{\closepart}{%
  \ifdefined\book@partopen
    \recordstat{part\thepart:problems}{\arabic{partproblems}}%
    \recordstat{part\thepart:chapters}{\arabic{partchapters}}%
  \fi}

\titlecontents{pchapter}[2.6em]{\addvspace{3pt}\normalsize}%
  {\contentslabel{2.2em}}{}{\titlerule*[0.7pc]{.}\contentspage}

\newcommand{\bookpart}[3]{%
  \closepart
  \part{#1}%
  \gdef\book@partopen{}%
  \begin{center}
  \begin{tcolorbox}[enhanced, colback=lightgray, colframe=primaryblue, boxrule=1.5pt,
      arc=3mm, width=0.85\textwidth, halign=center]
    {\Large\itshape\color{deepblue}#3\par}\medskip
    {\color{darkgray}#2\quad\textbullet\quad\stat{part\thepart:chapters} chapters\quad
     \textbullet\quad\stat{part\thepart:problems} problems}
  \end{tcolorbox}
  \end{center}
  \medskip
  {\large\bfseries\color{primaryblue}Chapters in this part\par}\smallskip
  \startcontents[parts]%
  \printcontents[parts]{p}{0}{\setcounter{tocdepth}{0}}}

% ---------------- Chapters ----------------
\newcommand{\bookchapter}[3][]{%
  \ifstrempty{#1}{\chapter{#2}}{\chapter[#1]{#2}}\label{ch:#3}%
  \gdef\currentchaptertitle{#2}%
  \stepcounter{partchapters}}

\newcommand{\chapterinfo}[2]{%
  \ifstrequal{#2}{Written}{\stepcounter{writtenchapters}}{}%
  \begin{center}\small\color{darkgray}
    \textbf{Level:}~#1\quad\textbullet\quad
    \textbf{Problems:}~\stat{ch\thechapter:problems}%
    \ifstrequal{#2}{Written}{~(\stat{ch\thechapter:solved} fully solved)}{~planned}%
    \quad\textbullet\quad\textbf{Status:}~#2
  \end{center}}

\newcommand{\closechapter}{%
  \recordstat{ch\thechapter:problems}{\arabic{chproblems}}%
  \recordstat{ch\thechapter:solved}{\arabic{chsolved}}}

\newcommand{\chaptersection}[1]{%
  \section*{#1}\addcontentsline{toc}{section}{#1}\markright{#1}}

% ---------------- Outline (not yet written) ----------------
\newcommand{\plannedtopic}[4][]{%
  \section{#2}%
  \countplanned{#3}%
  \begin{plannedbox}[Outline -- #3 problems planned]
    \textbf{Concepts to cover}
    \begin{itemize}[nosep, leftmargin=1.6em]
      #4
    \end{itemize}
    \ifstrempty{#1}{}{\par\smallskip\textbf{Note.}~#1}
  \end{plannedbox}}

\newcommand{\enrichmenttrack}{%
  \begin{tcolorbox}[enhanced, colback=warningred!6, colframe=warningred, boxrule=1pt, arc=2mm]
  \textbf{Enrichment track (not contest-rated).} Contest problems on this material are rare.
  When written, the chapter keeps the standard chapter tree, but its exercises are proofs and
  small computations rather than rated contest problems, and it carries no Codeforces band.
  \end{tcolorbox}}

\makeatother
