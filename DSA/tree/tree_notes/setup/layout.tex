% =====================================================================
%  setup/layout.tex -- page layout, headings, running heads, captions,
%  hyperlinks and PDF bookmarks.
% =====================================================================
\makeatletter

\setlength{\headheight}{15pt}
\raggedbottom                       % diagrams are non-floating; avoid stretched pages
\setcounter{secnumdepth}{2}         % chapter 1, topic 1.3, subsection 1.3.2
\setcounter{tocdepth}{1}            % printed TOC: parts, chapters, topics
\setlist{itemsep=0.15em, topsep=0.35em}

% ---------------- Part ----------------
% "top" class: the part page also carries the part overview written after \part.
\titleclass{\part}{top}
\titlespacing*{\part}{0pt}{40pt}{28pt}
\titleformat{\part}[display]
  {\centering\normalfont\Huge\bfseries\color{primaryblue}}
  {\partname\ \thepart}
  {20pt}
  {\Huge\color{deepblue}}
  [\vspace{2ex}{\color{primaryblue}\hrule height 3pt}\vspace{1ex}{\color{accentorange}\hrule height 1pt}]

% ---------------- Chapter ----------------
\titleformat{\chapter}[display]
  {\normalfont\huge\bfseries\color{primaryblue}}
  {\tikz{\node[circle, fill=primaryblue, text=white, minimum size=1.5cm, font=\Large\bfseries] {\thechapter};}\quad\chaptertitlename}
  {10pt}
  {\Huge\color{deepblue}}
  [\vspace{1ex}{\color{secondaryblue}\titlerule[2pt]}]
\titlespacing*{\chapter}{0pt}{-20pt}{24pt}

% ---------------- Section (= one topic) ----------------
\titleformat{\section}
  {\normalfont\Large\bfseries\color{primaryblue}}
  {\tikz{\filldraw[primaryblue] (0,0) rectangle (0.3,0.3);}\quad\thesection}
  {1em}
  {}
  [\vspace{0.5ex}{\color{lightgray}\titlerule[1pt]}]
\titlespacing*{\section}{0pt}{3ex plus 1ex minus .2ex}{1.4ex plus .2ex}

% ---------------- Subsection / subsubsection ----------------
\titleformat{\subsection}
  {\normalfont\large\bfseries\color{deepblue}}
  {\thesubsection}{0.8em}{}
\titlespacing*{\subsection}{0pt}{2.4ex plus 1ex minus .2ex}{1ex plus .2ex}
\titleformat{\subsubsection}
  {\normalfont\normalsize\bfseries\color{darkgray}}
  {}{0em}{}

% ---------------- Running heads ----------------
\renewcommand{\chaptermark}[1]{\markboth{\if@mainmatter\@chapapp~\thechapter.~\fi#1}{}}
\renewcommand{\sectionmark}[1]{\markright{\thesection~~#1}}
\pagestyle{fancy}
\fancyhf{}
\fancyhead[L]{\small\bfseries\color{primaryblue}\nouppercase{\leftmark}}
\fancyhead[R]{\small\color{darkgray}\nouppercase{\rightmark}}
\fancyfoot[L]{\footnotesize\itshape\color{darkgray}Trees for Competitive Programming}
\fancyfoot[R]{\small\color{darkgray}\thepage}
\renewcommand{\headrulewidth}{0.8pt}
\renewcommand{\footrulewidth}{0.4pt}
\renewcommand{\headrule}{\hbox to\headwidth{\color{primaryblue}\leaders\hrule height \headrulewidth\hfill}}
\renewcommand{\footrule}{\hbox to\headwidth{\color{lightgray}\leaders\hrule height \footrulewidth\hfill}}
\fancypagestyle{plain}{%
  \fancyhf{}%
  \fancyfoot[R]{\small\color{darkgray}\thepage}%
  \renewcommand{\headrulewidth}{0pt}%
  \renewcommand{\footrulewidth}{0pt}}

% ---------------- Captions and float numbering ----------------
\captionsetup{font=small, labelfont={bf,color=primaryblue}, margin=0.6cm, skip=5pt}
\counterwithin{figure}{section}     % Figure 1.3.2
\counterwithin{table}{section}      % Table 1.3.1

% ---------------- Hyperlinks and bookmarks ----------------
\hypersetup{
  colorlinks=true,
  linkcolor=primaryblue,
  urlcolor=secondaryblue,
  citecolor=accentorange,
  pdftitle={Trees for Competitive Programming},
  pdfauthor={Competitive Programming Guide}
}
\bookmarksetup{numbered, open, openlevel=1, depth=3}

\makeatother
