% =====================================================================
%  setup/code.tex -- one listing style for the whole book.
%
%    \begin{lstlisting}[caption={DFS on a tree}, label={lst:c01-dfs}]
%    ...
%    \end{lstlisting}                 -> "Code 1.3.2: DFS on a tree"
%
%  Rules: ASCII only inside listings; never inside a macro argument;
%  no \verb or verbatim anywhere in the book (use \code{} inline).
% =====================================================================
\lstdefinestyle{cpp}{
  language=C++,
  backgroundcolor=\color{backcolour},
  basicstyle=\ttfamily\small,
  keywordstyle=\color{codepurple}\bfseries,
  commentstyle=\color{codegray},
  stringstyle=\color{codegreen},
  numberstyle=\tiny\color{codegray},
  numbers=left, numbersep=8pt, xleftmargin=16pt,
  frame=lines, rulecolor=\color{codegray!40}, framesep=3pt,
  breaklines=true, breakatwhitespace=false,
  postbreak=\mbox{\textcolor{codegray}{$\hookrightarrow$}\space},
  showstringspaces=false, tabsize=4,
  columns=fullflexible, keepspaces=true, upquote=true,
  morekeywords={int64_t, uint64_t, __int128, size_t, constexpr, nullptr},
  captionpos=t, abovecaptionskip=3pt, belowcaptionskip=3pt,
  aboveskip=8pt, belowskip=8pt,
}
\lstset{style=cpp}
\renewcommand{\lstlistingname}{Code}
\AtBeginDocument{\counterwithin{lstlisting}{section}}

% Pseudocode (algpseudocode): comments in grey.
\algrenewcommand{\algorithmiccomment}[1]{\hfill{\color{codegray}$\triangleright$ #1}}
