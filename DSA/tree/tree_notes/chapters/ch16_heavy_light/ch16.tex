\bookchapter{Heavy--Light Decomposition}{c16}
\chapterinfo{CF 2000--2400}{Outline}

\plannedtopic{Splitting a Tree into Chains}{9}{%
  \item Heavy child (largest subtree) and light edges
  \item A path crosses at most $O(\log n)$ light edges
  \item Decomposing the tree into disjoint chains
  \item The head of each chain
}

\plannedtopic{HLD for Path Queries}{9}{%
  \item Assign a DFS-order position to every node
  \item A segment tree over the positions
  \item Path query and update by jumping chain to chain
  \item Combining with LCA
}

\plannedtopic{HLD Applications}{8}{%
  \item Path max/sum/assign with a lazy segment tree
  \item Subtree queries as one contiguous range
  \item Edge values on the deeper endpoint
}

\plannedtopic{HLD Practice}{7}{%
  \item Path queries on a tree (CSES / CF)
  \item Subtree and path updates together
  \item Debugging the chain decomposition on a small tree
}
