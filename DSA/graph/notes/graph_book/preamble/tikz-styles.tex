% ---------- macros ----------
\newcommand{\dparray}[4]{%
    \node[rowlabel] at (-0.15,#2) {#3};
    \foreach \v/\s [count=\i from 0] in {#4}{%
        \node[\s] (#1\i) at (0.45+\i*0.9,#2) {\v};
    }%
}
\newcommand{\dpindex}[5]{%
    \node[rowlabel, font=\scriptsize\color{darkgray}] at (-0.15,#2) {#3};
    \foreach \i [count=\c from 0] in {#4,...,#5}{%
        \node[idx] (#1\c) at (0.45+\c*0.9,#2) {\i};
    }%
}

% Legend chip:  \legend{style}{text}
\newcommand{\legendchip}[2]{%
    \tikz[baseline=-0.6ex]\node[#1, minimum width=5mm, minimum height=3.5mm]{};~#2}
